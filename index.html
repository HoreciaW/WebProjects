Google Calendar Free/Busy Slot Subtracter Logic

Runs inside your n8n JavaScript Code node. Takes studio business hours, subtracts booked calendar events, and filters available start times based on requested duration.


// ============================================================================
// N8N JAVASCRIPT CODE NODE: CALENDAR FREE/BUSY SLOT CALCULATION LOGIC
// ============================================================================

// 1. Extract requested appointment duration from incoming n8n item payload (defaulting to 120 mins if absent)
const requestedDuration = $json.requestedDurationMinutes || 120; 

// 2. Extract requested booking date string formatted as YYYY-MM-DD
const targetDateStr = $json.targetDate || "2026-10-20";

// 3. Extract existing busy calendar events array returned by the Google Calendar 'Get Events' node
const busyEvents = $json.gcalBusyEvents || [
  // Example array structure of existing appointments returned by Google Calendar:
  { start: `${targetDateStr}T11:00:00`, end: `${targetDateStr}T12:30:00` },
  { start: `${targetDateStr}T14:30:00`, end: `${targetDateStr}T15:30:00` }
];

// 4. Define Studio Operating Parameters
const BUSINESS_START_HOUR = 9;  // Studio opens at 9:00 AM
const BUSINESS_END_HOUR = 18;   // Studio closes at 6:00 PM (18:00)
const SLOT_INTERVAL_MINS = 30; // Check start time candidates every 30 minutes

// 5. Primary calculation function to compute open slots
function calculateAvailableSlots() {
  // Construct Date object for opening time on requested date
  const dayStart = new Date(`${targetDateStr}T0${BUSINESS_START_HOUR}:00:00`);
  
  // Construct Date object for closing time on requested date
  const dayEnd = new Date(`${targetDateStr}T${BUSINESS_END_HOUR}:00:00`);
  
  // Parse GCal ISO string dates into JavaScript Date instances for numeric comparison
  const busyIntervals = busyEvents.map(e => ({
    start: new Date(e.start), // Start timestamp of existing booking
    end: new Date(e.end)       // End timestamp of existing booking
  }));

  // Initialize array to hold valid non-overlapping slot candidates
  const validSlots = [];
  
  // Pointer initialized to studio opening time
  let currentCandidate = new Date(dayStart);

  // Loop through operating hours in 30-minute steps
  while (currentCandidate < dayEnd) {
    // Calculate required finish time by adding requested appointment duration (in milliseconds)
    const candidateEnd = new Date(currentCandidate.getTime() + requestedDuration * 60000);

    // Rule A: Skip candidate if appointment extends past studio closing time
    if (candidateEnd > dayEnd) {
      // Step forward by 30-minute interval
      currentCandidate = new Date(currentCandidate.getTime() + SLOT_INTERVAL_MINS * 60000);
      continue; // Skip to next candidate
    }

    // Rule B: Check if candidate window overlaps with ANY booked calendar event
    const hasOverlap = busyIntervals.some(busy => {
      // Standard time-range overlap formula: (StartA < EndB) AND (EndA > StartB)
      return currentCandidate < busy.end && candidateEnd > busy.start;
    });

    // Rule C: If no conflict detected, record this valid start slot
    if (!hasOverlap) {
      validSlots.push({
        // Format readable time string for client UI (e.g., "10:00 AM")
        formattedTime: currentCandidate.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        // Provide exact ISO timestamps for Google Calendar event creation node
        isoStart: currentCandidate.toISOString(),
        isoEnd: candidateEnd.toISOString()
      });
    }

    // Step forward by interval (30 mins) to check next start time slot candidate
    currentCandidate = new Date(currentCandidate.getTime() + SLOT_INTERVAL_MINS * 60000);
  }

  // Return final list of valid appointment start times
  return validSlots;
}

// 6. Return structured data object for n8n workflow execution pipeline
return [{
  json: {
    status: "success",                           // Operation status
    targetDate: targetDateStr,                  // Booking date
    totalDurationMinutes: requestedDuration,    // Appointment length
    availableSlotsCount: calculateAvailableSlots().length, // Total available options
    availableSlots: calculateAvailableSlots()   // Array of open time slots
  }
}];
